Create `test/docker-compose.test.yml` with exactly this structure:

services:
  app:
    build:
      context: ..
      dockerfile: Dockerfile
    environment:
      - LLM_MOCK=true
      - OPENROUTER_API_KEY=test-key
    volumes:
      - test-db:/app/db
    networks:
      - test-net
    healthcheck:
      test: ["CMD-SHELL", "curl -f http://localhost:8000/api/health || exit 1"]
      interval: 5s
      timeout: 3s
      retries: 12
      start_period: 30s

  playwright:
    image: mcr.microsoft.com/playwright:v1.52.0-noble
    working_dir: /tests
    volumes:
      - ./:/tests
    environment:
      - BASE_URL=http://app:8000
    command: sh -c "npm install --quiet && npx playwright test"
    depends_on:
      app:
        condition: service_healthy
    networks:
      - test-net

networks:
  test-net:
    driver: bridge

volumes:
  test-db:

Key decisions implemented:
- D-07: Isolated bridge network `test-net`; playwright uses `http://app:8000` not localhost
- D-08: `LLM_MOCK=true` + `OPENROUTER_API_KEY=test-key` (dummy; no real key needed)
- D-09: volume `test-db:` has no `name:` field — gets project-prefixed name (not "finally-data")
- D-10: `depends_on: app: condition: service_healthy` ensures playwright waits for backend readiness
- `build: context: ..` because this file is in `test/` but Dockerfile is at project root
- `volumes: - ./:/tests` mounts the `test/` directory into the container at `/tests`
- `npm install --quiet && npx playwright test` installs @playwright/test at runtime, then runs tests
- `start_period: 30s` + `retries: 12` gives the app 60+ seconds to start (Docker build + DB init)
  </action>
