```typescript
interface PriceUpdate {
  ticker: string;
  price: number;
  previous_price: number;
  timestamp: number;
  change: number;
  change_percent: number;
  direction: 'up' | 'down' | 'flat';
}

interface PriceStore {
  prices: Record<string, PriceUpdate>;
  status: 'CONNECTING' | 'OPEN' | 'CLOSED';
  setPrices(updates: Record<string, PriceUpdate>): void;
  setStatus(status: PriceStore['status']): void;
}

export declare const usePriceStore: UseBoundStore<StoreApi<PriceStore>>;
```

    Create frontend/src/components/Sparkline.tsx (per D-02 — pure SVG, no library):

    Props: `{ prices: number[]; width?: number; height?: number }`
    Defaults: width=80, height=32

    Implementation:
    - If prices has fewer than 2 points, render an empty `<svg>` with the given dimensions
    - Compute min and max of the prices array; if min === max, add/subtract 0.5 to avoid division by zero
    - Map each price to (x, y) coordinates:
      - x = (index / (prices.length - 1)) * width
      - y = height - ((price - min) / (max - min)) * height (invert Y so higher = up)
    - Render a single `<polyline>` with `points` attribute as space-joined "x,y" strings
    - Style: `fill="none"`, `stroke="#209dd7"` (accent-blue), `strokeWidth="1.5"`
    - SVG viewBox: `"0 0 {width} {height}"`, no overflow

    Export: `export function Sparkline({ prices, width = 80, height = 32 }: SparklineProps)`

    Create frontend/src/components/WatchlistPanel.tsx as a client component:

    Props interface:
    ```
    interface WatchlistPanelProps {
      selectedTicker: string | null;
      onSelectTicker: (ticker: string) => void;
      onTradeOpen?: (ticker: string) => void;  // optional, for pre-filling trade bar
    }
    ```

    State:
    - `watchlist: string[]` — list of tickers from /api/watchlist (ticker strings only)
    - `sparklineHistory: Record<string, number[]>` — price history per ticker accumulated from SSE

    On mount: fetch `/api/watchlist`, parse response as WatchlistResponse, set `watchlist` to the ticker strings.

    SSE integration (per D-07, D-09):
    - Subscribe to `usePriceStore(state => state.prices)` — this gives the full price map on every SSE update
    - In a `useEffect` that depends on `prices`, for each ticker in `watchlist`:
      - Look up `prices[ticker]`; if present, append `prices[ticker].price` to `sparklineHistory[ticker]`
      - Keep only the last 60 points per ticker (`.slice(-60)`)
    - Use functional state update: `setSparklineHistory(prev => { ... })`

    Price flash (per D-10):
    - Maintain `flashState: Record<string, 'up' | 'down' | null>` state
    - When `prices` changes, compare new `prices[ticker].price` vs `prices[ticker].previous_price`:
      - If direction is 'up': set flashState[ticker] = 'up', schedule setTimeout 500ms to clear it back to null
      - If direction is 'down': set flashState[ticker] = 'down', schedule setTimeout 500ms to clear it back to null
    - Apply CSS class `flash-up` or `flash-down` to the price cell `<td>` based on flashState[ticker]
    - Clear timeouts on cleanup to avoid memory leaks (store timeout IDs in a ref)

    Watchlist management buttons:
    - Each row has a small "x" button (text-gray-400 hover:text-red-400) that calls `DELETE /api/watchlist/{ticker}` then re-fetches watchlist
    - An input + "Add" button at the bottom of the panel: on submit, calls `POST /api/watchlist` with `{ticker: input.toUpperCase()}`, then re-fetches watchlist on success (ignore 400 if already exists, show nothing)

    Table columns: Ticker | Price | Change% | Sparkline | (remove button)
    - Ticker: bold, accent-yellow text, cursor-pointer on row click
    - Price: monospace, right-aligned, flash class applied here
    - Change%: green text if positive, red text if negative (use inline style or conditional Tailwind)
    - Sparkline: render `<Sparkline prices={sparklineHistory[ticker] ?? []} />`
    - Row click: calls `onSelectTicker(ticker)`; selected row gets subtle bg-[#1e2a3a] highlight

    Styling: dark table (`bg-[#0d1117]`), thin gray borders (`border-[#2a2a3a]`), small font (text-xs or text-sm). Full height of left column with `overflow-y: auto`.

    Export: `export function WatchlistPanel({ selectedTicker, onSelectTicker }: WatchlistPanelProps)`
